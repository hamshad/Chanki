// src/data/device-id.ts
import { db } from './db'

export async function getOrCreateDeviceId(): Promise<string> {
  const existing = await db.meta.get('deviceId')
  if (existing) return existing.value as string

  const deviceId = crypto.randomUUID()
  await db.meta.put({ key: 'deviceId', value: deviceId })
  return deviceId
}
```

Requirements:
- Uses native `crypto.randomUUID()` (no dependencies)
- Stores in `meta` table with key `'deviceId'`
- Returns existing value on subsequent calls (idempotent)
- No login UI, no fingerprinting, no PII
- Throws on crypto unavailable (should not happen in targets)

// src/data/seed.ts
import starterDeck from '/assets/deck/hsk1-starter.json' // Vite import
import { db } from './db'
import { CardSchema, validateCard } from './schema'
import type { Deck, Card } from './schema'

export async function seedStarterDeckIfNeeded(): Promise<void> {
  const seeded = await db.meta.get('starterDeckSeeded')
  if (seeded) return

  await db.transaction('rw', db.decks, db.cards, async () => {
    // Insert deck
    await db.decks.put(starterDeck.deck as Deck)

    // Validate and insert cards
    const validCards = starterDeck.cards
      .map((c: unknown) => validateCard({ ...c, schemaVersion: 1, createdAt: Date.now(), updatedAt: Date.now() }))
    await db.cards.bulkAdd(validCards as Card[])

    await db.meta.put({ key: 'starterDeckSeeded', value: true })
  })
}
```

Requirements:
- Imports starter deck JSON via Vite (`/assets/deck/hsk1-starter.json`)
- Checks `meta.starterDeckSeeded` first — idempotent
- Uses `db.transaction('rw', db.decks, db.cards, ...)` for atomicity
- Validates each card via `validateCard` (Zod parse) — rejects invalid
- Adds `schemaVersion: 1`, `createdAt`, `updatedAt` to each card
- Uses `bulkAdd` for performance
- Marks `starterDeckSeeded: true` in meta on success

Run all tests to verify integration.
